cmake_minimum_required(VERSION 3.24)
project(effetune_audio_engine LANGUAGES C CXX)

option(ET_AUDIO_ENGINE_X64_V3 "Build the AVX2 and FMA executable" OFF)
option(ET_AUDIO_ENGINE_BUILD_TESTS "Build native audio engine tests" ON)
option(ET_AUDIO_ENGINE_OPTIMIZE_FP "Allow equivalent floating-point transformations" ON)
option(ET_AUDIO_ENGINE_HOST_ONLY "Build only the reusable native host library" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  # Match the shipped STL layout in Debug and Release; the DSP kernel slots
  # have fixed size limits. The DSP allocation guard stays enabled in Debug.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
set(dsp_root "${CMAKE_CURRENT_SOURCE_DIR}/../../dsp")
set(openhome_src "${CMAKE_CURRENT_SOURCE_DIR}/../openhome-sidecar/src")

if(NOT ET_AUDIO_ENGINE_HOST_ONLY)
  # Both main and its CPU check must be compiled for the baseline ISA.
  add_library(audio_engine_entry OBJECT app/entry.cpp)
  if(ET_AUDIO_ENGINE_X64_V3)
    target_compile_definitions(audio_engine_entry PRIVATE ET_AUDIO_ENGINE_X64_V3=1)
    add_compile_options(-mavx2 -mfma)
    set(engine_suffix "-v3")
  endif()
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(ET_SIMD ON CACHE BOOL "" FORCE)
endif()
if(ET_AUDIO_ENGINE_OPTIMIZE_FP AND NOT ET_AUDIO_ENGINE_HOST_ONLY)
  foreach(flag -fno-signed-zeros -fno-trapping-math -fassociative-math -freciprocal-math -ffp-contract=fast)
    if(MSVC)
      list(APPEND audio_engine_fp "/clang:${flag}")
    else()
      list(APPEND audio_engine_fp "${flag}")
    endif()
  endforeach()
endif()
if(NOT TARGET effetune_dsp_core)
  add_subdirectory("${dsp_root}" dsp)
endif()
if(audio_engine_fp)
  get_directory_property(dsp_targets DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/dsp" BUILDSYSTEM_TARGETS)
  foreach(target IN LISTS dsp_targets)
    get_target_property(kind ${target} TYPE)
    if(NOT kind STREQUAL "INTERFACE_LIBRARY" AND NOT kind STREQUAL "UTILITY")
      target_compile_options(${target} PRIVATE ${audio_engine_fp})
    endif()
  endforeach()
  get_target_property(core_sources effetune_dsp_core SOURCES)
  foreach(source IN LISTS core_sources)
    cmake_path(ABSOLUTE_PATH source BASE_DIRECTORY "${dsp_root}" NORMALIZE)
    get_source_file_property(flags "${source}" TARGET_DIRECTORY effetune_dsp_core COMPILE_OPTIONS)
    if(flags)
      list(FILTER flags EXCLUDE REGEX "(ffp-contract=off|/fp:strict)")
      set_property(SOURCE "${source}" TARGET_DIRECTORY effetune_dsp_core PROPERTY COMPILE_OPTIONS "${flags}")
    endif()
  endforeach()
endif()
if(MSVC)
  set(audio_engine_warnings /W4 /WX /utf-8 /EHsc)
else()
  set(audio_engine_warnings -Wall -Wextra -Wpedantic -Werror)
endif()
if(NOT ET_AUDIO_ENGINE_HOST_ONLY AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_link_options(-static-libstdc++ -static-libgcc)
endif()
add_library(audio_engine_host STATIC
  host/session.cpp host/pipeline.cpp host/assets.cpp host/topology.cpp host/output_stage.cpp host/midi.cpp host/telemetry.cpp host/power.cpp
  host/clocked_fifo.cpp host/file_fifo.cpp host/resampler.cpp host/io_pipeline.cpp host/network_output.cpp
  host/visualizer.cpp host/spectrum.cpp host/frequency_preview.cpp
  device/null/null_device.cpp)
target_compile_options(audio_engine_host PRIVATE ${audio_engine_fp} ${audio_engine_warnings})
target_include_directories(audio_engine_host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${dsp_root}/core")
target_link_libraries(audio_engine_host PUBLIC effetune_dsp_core)
if(ET_AUDIO_ENGINE_HOST_ONLY)
  return()
endif()

add_library(audio_engine_protocol STATIC app/frame.cpp "${openhome_src}/protocol.cpp")
target_compile_options(audio_engine_protocol PRIVATE ${audio_engine_fp} ${audio_engine_warnings})
target_include_directories(audio_engine_protocol PUBLIC app "${openhome_src}")
find_package(Threads REQUIRED)
target_link_libraries(audio_engine_protocol PRIVATE Threads::Threads)
if(WIN32)
  set(platform_source app/platform_win.cpp)
else()
  set(platform_source app/platform_posix.cpp)
endif()
add_executable(effetune_audio_engine app/sidecar.cpp app/device_host.cpp app/sampler_control.cpp app/output_control.cpp app/midi_control.cpp app/network_reports.cpp ${platform_source} $<TARGET_OBJECTS:audio_engine_entry>)
target_compile_options(effetune_audio_engine PRIVATE ${audio_engine_fp} ${audio_engine_warnings})
set_target_properties(effetune_audio_engine PROPERTIES OUTPUT_NAME "effetune-audio-engine${engine_suffix}")
target_sources(effetune_audio_engine PRIVATE app/power_control.cpp)
target_sources(effetune_audio_engine PRIVATE app/measurement.cpp app/measurement_control.cpp)
target_sources(effetune_audio_engine PRIVATE app/instrument_control.cpp)
target_sources(effetune_audio_engine PRIVATE device/null/realtime_null.cpp)
target_link_libraries(effetune_audio_engine PRIVATE audio_engine_host audio_engine_protocol)
set(miniaudio_sources device/miniaudio/miniaudio_device.cpp device/miniaudio/miniaudio_impl.cpp)
if(MSVC)
  set_source_files_properties(device/miniaudio/miniaudio_impl.cpp PROPERTIES COMPILE_OPTIONS /w)
else()
  set_source_files_properties(device/miniaudio/miniaudio_impl.cpp PROPERTIES COMPILE_OPTIONS -w)
endif()
if(WIN32)
  target_sources(effetune_audio_engine PRIVATE device/wasapi/wasapi_device.cpp)
  target_compile_definitions(effetune_audio_engine PRIVATE ET_AUDIO_ENGINE_WASAPI=1)
  target_link_libraries(effetune_audio_engine PRIVATE ole32 avrt uuid ksuser)
else()
  target_sources(effetune_audio_engine PRIVATE ${miniaudio_sources})
  target_compile_definitions(effetune_audio_engine PRIVATE ET_AUDIO_ENGINE_MINIAUDIO=1)
  if(APPLE)
    target_link_libraries(effetune_audio_engine PRIVATE "-framework CoreAudio" "-framework AudioToolbox"
                          "-framework CoreFoundation")
  else()
    target_link_libraries(effetune_audio_engine PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
  endif()
endif()
if(ET_AUDIO_ENGINE_BUILD_TESTS)
  enable_testing()
  add_executable(audio_engine_tests tests/audio_engine_tests.cpp)
  target_compile_options(audio_engine_tests PRIVATE ${audio_engine_fp} ${audio_engine_warnings})
  set_target_properties(audio_engine_tests PROPERTIES OUTPUT_NAME "audio-engine-tests${engine_suffix}")
  target_link_libraries(audio_engine_tests PRIVATE audio_engine_host audio_engine_protocol)
  add_test(NAME audio_engine_tests COMMAND audio_engine_tests)
  foreach(test rate_conversion_tests device_signal_tests)
    add_executable(${test} tests/${test}.cpp)
    target_compile_options(${test} PRIVATE ${audio_engine_fp} ${audio_engine_warnings})
    set_target_properties(${test} PROPERTIES OUTPUT_NAME "${test}${engine_suffix}")
    target_link_libraries(${test} PRIVATE audio_engine_host)
    add_test(NAME ${test} COMMAND ${test})
  endforeach()
  add_executable(device_backend_tests tests/device_backend_tests.cpp ${miniaudio_sources})
  target_compile_options(device_backend_tests PRIVATE ${audio_engine_fp} ${audio_engine_warnings})
  target_include_directories(device_backend_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(device_backend_tests PRIVATE ET_AUDIO_ENGINE_TEST_MINIAUDIO_NULL=1)
  target_link_libraries(device_backend_tests PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  if(UNIX AND NOT APPLE)
    target_link_libraries(device_backend_tests PRIVATE m)
  endif()
  set_target_properties(device_backend_tests PROPERTIES OUTPUT_NAME "device-backend-tests${engine_suffix}")
  add_test(NAME device_backend_tests COMMAND device_backend_tests)
endif()
