cmake_minimum_required(VERSION 3.24)
project(effetune_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
get_filename_component(EFFETUNE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_subdirectory("${EFFETUNE_ROOT}/dsp" "${CMAKE_CURRENT_BINARY_DIR}/dsp")
find_package(oboe REQUIRED CONFIG)
add_library(effetune_android SHARED native_engine.cpp jni_bridge.cpp finite_capture.cpp pcm_resampler.cpp pcm_resampler_jni.cpp)
target_include_directories(effetune_android PRIVATE "${EFFETUNE_ROOT}/dsp/core")
target_link_libraries(effetune_android PRIVATE effetune_dsp_core oboe::oboe android log aaudio)
target_compile_options(effetune_android PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(effetune_android PRIVATE "-Wl,-z,max-page-size=16384")

add_executable(android_runtime_test EXCLUDE_FROM_ALL native_engine_test.cpp native_engine.cpp finite_capture.cpp pcm_resampler.cpp)
target_include_directories(android_runtime_test PRIVATE "${EFFETUNE_ROOT}/dsp/core" "${EFFETUNE_ROOT}/dsp/generated/cpp")
target_link_libraries(android_runtime_test PRIVATE effetune_dsp_core oboe::oboe android log aaudio)
target_compile_options(android_runtime_test PRIVATE -Wall -Wextra -Wpedantic -Werror)

add_executable(android_dsp_parity EXCLUDE_FROM_ALL "${EFFETUNE_ROOT}/dsp/test/parity_runner.cpp")
target_include_directories(android_dsp_parity PRIVATE "${EFFETUNE_ROOT}/dsp/core")
target_link_libraries(android_dsp_parity PRIVATE effetune_dsp_core)
